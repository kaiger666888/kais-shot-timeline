```python
def _producer_registry_integrity(asset_dir: Path) -> list:
    """Phase 7: registry↔shots cross-file integrity for PRODUCER asset dir.
    
    Gated on file existence: when characters.json/props.json/registry.draft.json
    are absent (v1.0 asset / route-down degrade), returns [] (clean — no-op).
    When present, checks:
      (a) appearance_shots[] in characters/props ⊆ shots.json IDs
      (b) registry clusters[].members[].shot_id ⊆ shots.json IDs
      (c) canonical IDs unique + match ^(char|prop)_[0-9]{3}$
      (d) NO review_state:"proposed" in canonical files (Pitfall 7 second-line assert)
    """
    import re
    failures = []
    shots_path = asset_dir / "shots.json"
    if not shots_path.is_file():
        return []   # can't check without shots.json
    try:
        shots = json.loads(shots_path.read_text(encoding="utf-8"))
        shot_ids = {s.get("id") for s in shots} if isinstance(shots, list) else set()
    except (json.JSONDecodeError, TypeError):
        return [f"shots.json unreadable in {asset_dir}"]
    
    # Check each canonical file that exists
    for name, id_prefix in (("characters.json", "char_"), ("props.json", "prop_")):
        path = asset_dir / name
        if not path.is_file():
            continue   # absent = graceful-degrade (v1.0 asset)
        try:
            entries = json.loads(path.read_text(encoding="utf-8"))
        except json.JSONDecodeError as e:
            failures.append(f"{name}: invalid JSON: {e}")
            continue
        if not isinstance(entries, list):
            failures.append(f"{name}: expected array, got {type(entries).__name__}")
            continue
        seen_ids = set()
        for entry in entries:
            eid = entry.get("id")
            # (c) ID pattern
            if not (isinstance(eid, str) and re.match(rf"^{id_prefix}[0-9]{{3}}$", eid)):
                failures.append(f"{name}: ID {eid!r} does not match {id_prefix}_[0-9]{{3}}")
            # (c) ID uniqueness
            if eid in seen_ids:
                failures.append(f"{name}: duplicate ID {eid!r}")
            seen_ids.add(eid)
            # (d) Pitfall 7 — NO proposed in canonical
            if entry.get("review_state") == "proposed":
                failures.append(f"{name} {eid}: review_state='proposed' leaked into canonical (Pitfall 7)")
            # (a) appearance_shots ⊆ shots
            for sid in entry.get("appearance_shots", []) or []:
                if sid not in shot_ids:
                    failures.append(f"{name} {eid}: appearance_shots references unknown shot {sid}")
    
    # (b) registry cluster members ⊆ shots.json
    reg_path = asset_dir / "registry.draft.json"
    if reg_path.is_file():
        try:
            registry = json.loads(reg_path.read_text(encoding="utf-8"))
            if isinstance(registry, dict):
                for cl in registry.get("clusters", []):
                    cid = cl.get("cluster_id", "?")
                    for m in cl.get("members", []) or []:
                        if m.get("shot_id") not in shot_ids:
                            failures.append(f"registry cluster {cid}: member shot_id {m.get('shot_id')} unknown")
        except json.JSONDecodeError as e:
            failures.append(f"registry.draft.json: invalid JSON: {e}")
    
    return failures
```
