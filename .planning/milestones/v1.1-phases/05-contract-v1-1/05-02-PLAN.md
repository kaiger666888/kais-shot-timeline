```python
def _git_sha() -> str:
    """取仓库短 git SHA 作为 generator.version；失败回退 "dev"。"""
    try:
        r = subprocess.run(
            ["git", "-C", str(REPO), "rev-parse", "--short", "HEAD"],
            capture_output=True, text=True, timeout=2)
        return r.stdout.strip() or "dev"
    except (subprocess.SubprocessError, OSError):
        return "dev"
```
